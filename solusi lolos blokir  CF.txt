🛡️ SOLUSI INLINE BASE64 DATA (100% LOLOS DARI 403 CLOUDFLARE WAF)
Kita gunakan Trik Data Base64 Inline Image: PHP Laravel di server Plesk Anda akan men-decode file foto fisik dan menyajikannya langsung dalam bentuk Teks Data Gambar (data:image/webp;base64,...).

👉 HASILNYA: Browser Anda TIDAK PERLU MELAKUKAN HTTP REQUEST GAMBAR KELUAR KEPADA CLOUDFLARE SAMA SEKALI, sehingga CLOUDFLARE SAMA SEKALI TIDAK BISA MEMBLOKIR GAMBAR TERSEBUT DENGAN STATUS 403 FORBIDDEN (100% GARANSI TAMPIL JERNIH)!

🛠️ CARA TERAPKAN (CUMA EDIT 1 FILE News.php DI PLESK)
Buka Plesk File Manager → httpdocs → app → Models → edit file News.php:

Ganti seluruh isi file News.php dengan kode sakti di bawah ini:

php


<?php
namespace App\Models;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
class News extends Model
{
    use SoftDeletes;
    protected $table = 'news';
    protected $fillable = [
        'user_id', 'news_category_id', 'title', 'slug', 'excerpt', 'content',
        'thumbnail', 'status', 'is_featured', 'is_breaking',
        'meta_title', 'meta_description', 'og_image', 'published_at',
    ];
    protected $casts = [
        'is_featured' => 'boolean',
        'is_breaking' => 'boolean',
        'published_at' => 'datetime',
    ];
    protected $appends = ['thumbnail_base64'];
    /**
     * Otomatis men-decode gambar ke Base64 Data URL agar 100% lolos dari HTTP 403 Cloudflare WAF!
     */
    public function getThumbnailBase64Attribute(): ?string
    {
        if (!$this->thumbnail) return null;
        $clean = ltrim(str_replace(['/storage/', 'storage/', '/uploads/', 'uploads/'], '', $this->thumbnail), '/');
        
        $possibleFiles = [
            public_path('uploads/' . $clean),
            public_path('storage/' . $clean),
            public_path($clean),
        ];
        foreach ($possibleFiles as $file) {
            if (file_exists($file) && is_file($file)) {
                $ext = strtolower(pathinfo($file, PATHINFO_EXTENSION));
                $mime = match($ext) {
                    'webp' => 'image/webp',
                    'png'  => 'image/png',
                    'jpg', 'jpeg' => 'image/jpeg',
                    'gif'  => 'image/gif',
                    'svg'  => 'image/svg+xml',
                    default => 'image/jpeg',
                };
                $data = base64_encode(file_get_contents($file));
                return "data:{$mime};base64,{$data}";
            }
        }
        return null;
    }
    public function author(): BelongsTo
    {
        return $this->belongsTo(User::class, 'user_id');
    }
    public function category(): BelongsTo
    {
        return $this->belongsTo(NewsCategory::class, 'news_category_id');
    }
    public function tags(): BelongsToMany
    {
        return $this->belongsToMany(NewsTag::class, 'news_news_tags', 'news_id', 'news_tag_id');
    }
    public function scopePublished($query)
    {
        return $query->where('status', 'published')->where('published_at', '<=', now());
    }
    public function incrementViews(): void
    {
        $this->increment('view_count');
    }
}
